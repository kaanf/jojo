import subprocess, os, signal, sys, collections
n=int(sys.argv[1]); c=collections.Counter()
for i in range(n):
    p=subprocess.Popen(['./child'],stdin=subprocess.PIPE,stdout=subprocess.PIPE)
    p.stdout.readline()
    os.kill(p.pid, signal.SIGKILL)
    p.stdin.close()
    c[p.wait()]+=1
print(dict(c))
