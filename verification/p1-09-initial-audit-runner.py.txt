import sys, json, re
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'scripts'))
import verify_p1_04 as shared
def inventory(output, passed):
    names=[]
    for path in ['crates/jojo-domain/tests/t09_streams.rs','crates/jojo-storage/tests/t09_process.rs']:
        names += re.findall(r'#\[test\]\s*fn (\w+)\(', (ROOT/path).read_text())
    missing=sorted(set(names)-set(passed))
    if missing: raise RuntimeError(f'Missing T09 tests: {missing}')
    return dict(t09_total=len(names),t09_test_names=names)
try:
    shared.main(task='P1-09',extra=inventory,scope='P1-09 completion audit; local fixtures only')
finally:
    path=ROOT/'verification/p1-09-result.json'
    result=json.loads(path.read_text())
    result.pop('p1_05',None)
    result['p1_10']='NO-GO'
    result['acceptance_complete']=False
    result['audit_findings']=['PID identity inspection and signal are not atomic; ps start identity has one-second resolution.', 'Vigla source attribution/license and REPOSITORY_PINS entry missing.', 'Authoritative P1-09 completion runner/report were absent at audit start.']
    path.write_text(json.dumps(result,indent=2)+'\n')
